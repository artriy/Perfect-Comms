cmake_minimum_required(VERSION 3.24)
project(perfect-comms-obs VERSION 1.0.0 LANGUAGES CXX)

if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Perfect Comms OBS currently supports Windows x64 only")
endif()
set(OBS_SOURCE_DIR "" CACHE PATH "Official obs-studio 32.2.2 source root")
set(OBS_LIBRARY "" CACHE FILEPATH "Official obs.lib (or import library generated from official obs.dll)")
set(OBS_FRONTEND_LIBRARY "" CACHE FILEPATH "Official obs-frontend-api.lib")
foreach(required OBS_SOURCE_DIR OBS_LIBRARY OBS_FRONTEND_LIBRARY)
  if(NOT EXISTS "${${required}}")
    message(FATAL_ERROR "Set ${required} to an existing official OBS 32.2.2 SDK path")
  endif()
endforeach()
if(NOT EXISTS "${OBS_SOURCE_DIR}/libobs/obs-module.h" OR NOT EXISTS "${OBS_SOURCE_DIR}/frontend/api/obs-frontend-api.h")
  message(FATAL_ERROR "OBS_SOURCE_DIR must contain official libobs and frontend/api headers")
endif()
file(READ "${OBS_SOURCE_DIR}/libobs/obs-config.h" obs_version_header)
if(NOT obs_version_header MATCHES "LIBOBS_API_MAJOR_VER 32" OR NOT obs_version_header MATCHES "LIBOBS_API_MINOR_VER 2" OR NOT obs_version_header MATCHES "LIBOBS_API_PATCH_VER 2")
  message(FATAL_ERROR "This plugin targets official OBS 32.2.2 headers")
endif()
set(OBS_DATA_PATH "data")
set(OBS_PLUGIN_PATH "obs-plugins")
set(OBS_PLUGIN_DESTINATION "obs-plugins/64bit")
set(OBS_RELEASE_CANDIDATE 0)
set(OBS_BETA 0)
configure_file("${OBS_SOURCE_DIR}/libobs/obsconfig.h.in" "${CMAKE_CURRENT_BINARY_DIR}/sdk/obsconfig.h" @ONLY)

add_library(perfect-comms-recording STATIC recording.cpp timing.cpp)
target_compile_features(perfect-comms-recording PUBLIC cxx_std_20)
target_link_libraries(perfect-comms-recording PUBLIC ntdll)
target_include_directories(perfect-comms-recording PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(perfect-comms-recording PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0602)
set_target_properties(perfect-comms-recording PROPERTIES POSITION_INDEPENDENT_CODE ON)

add_library(perfect-comms-obs MODULE plugin.cpp bridge.cpp)
target_compile_features(perfect-comms-obs PRIVATE cxx_std_20)
target_include_directories(perfect-comms-obs PRIVATE
  "${OBS_SOURCE_DIR}/libobs"
  "${OBS_SOURCE_DIR}/frontend/api"
  "${CMAKE_CURRENT_BINARY_DIR}/sdk")
target_link_libraries(perfect-comms-obs PRIVATE perfect-comms-recording "${OBS_LIBRARY}" "${OBS_FRONTEND_LIBRARY}" advapi32)
set_target_properties(perfect-comms-obs PROPERTIES PREFIX "" OUTPUT_NAME "perfect-comms-obs")
if(MSVC)
  target_compile_options(perfect-comms-recording PRIVATE /W4 /permissive- /EHsc /utf-8)
  target_compile_options(perfect-comms-obs PRIVATE /W4 /permissive- /EHsc /utf-8)
else()
  target_compile_options(perfect-comms-recording PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(perfect-comms-obs PRIVATE -Wall -Wextra -Wpedantic)
endif()

install(TARGETS perfect-comms-obs RUNTIME DESTINATION obs-plugins/64bit LIBRARY DESTINATION obs-plugins/64bit)
include(CTest)
if(BUILD_TESTING)
  add_executable(perfect-comms-recording-tests tests/recording-tests.cpp)
  target_link_libraries(perfect-comms-recording-tests PRIVATE perfect-comms-recording)
  add_test(NAME perfect-comms-recording-behavior COMMAND perfect-comms-recording-tests)
endif()
